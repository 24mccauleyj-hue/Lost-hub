local function draggable(frame,handle)

	local dragging = false
	local start
	local position

	handle.Active = true

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			start = input.Position
			position = frame.Position
		end
	end)

	UserInputService.InputChanged:Connect(function(input)

		if not dragging then return end

		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then

			local delta = input.Position - start

			frame.Position = UDim2.new(
				position.X.Scale,
				position.X.Offset + delta.X,
				position.Y.Scale,
				position.Y.Offset + delta.Y
			)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)

		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			dragging = false
		end
	end)
end

RunService.RenderStepped:Connect(function()

	if not Fly or not Root then return end

	local camera = workspace.CurrentCamera
	local direction = Vector3.zero

	if UserInputService:IsKeyDown(Enum.KeyCode.W) then
		direction += camera.CFrame.LookVector
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.S) then
		direction -= camera.CFrame.LookVector
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.A) then
		direction -= camera.CFrame.RightVector
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.D) then
		direction += camera.CFrame.RightVector
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
		direction += Vector3.new(0,1,0)
	end

	if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
		direction -= Vector3.new(0,1,0)
	end

	if direction.Magnitude > 0 then
		direction = direction.Unit
	end

	Root.AssemblyLinearVelocity = direction * FlySpeed
end)

RunService.Heartbeat:Connect(function()

	if not AntiFling or not Root or not Humanoid then
		return
	end

	-- Prevent extreme local velocity spikes.
	local velocity = Root.AssemblyLinearVelocity

	if velocity.Magnitude > 120 then
		Root.AssemblyLinearVelocity =
			Vector3.new(0,math.clamp(velocity.Y,-50,50),0)
	end

	-- Recover from local ragdoll/falling states.
	local state = Humanoid:GetState()

	if state == Enum.HumanoidStateType.Ragdoll
		or state == Enum.HumanoidStateType.FallingDown then

		Humanoid:ChangeState(
			Enum.HumanoidStateType.GettingUp
		)
	end
end)

local function removeESP(player)

	if ESPObjects[player] then

		for _,obj in ipairs(ESPObjects[player]) do
			if obj then
				obj:Destroy()
			end
		end

		ESPObjects[player] = nil
	end
end

local function addESP(player)

	if player == Player or not ESP then
		return
	end

	local char = player.Character
	if not char then return end

	removeESP(player)

	local highlight = Instance.new("Highlight")
	highlight.Adornee = char
	highlight.FillColor = RED
	highlight.OutlineColor = RED
	highlight.FillTransparency = .8
	highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	highlight.Parent = Gui

	local tag

	local head = char:FindFirstChild("Head")

	if head then

		tag = Instance.new("BillboardGui")
		tag.Adornee = head
		tag.Size = UDim2.fromOffset(160,30)
		tag.StudsOffset = Vector3.new(0,2.5,0)
		tag.AlwaysOnTop = true
		tag.Parent = Gui

		local text = Instance.new("TextLabel")
		text.Size = UDim2.fromScale(1,1)
		text.BackgroundTransparency = 1
		text.Text = player.DisplayName
		text.TextColor3 = RED
		text.TextStrokeTransparency = 0
		text.Font = Enum.Font.Code
		text.TextSize = 13
		text.Parent = tag
	end

	ESPObjects[player] = {highlight,tag}
end

ESPButton.MouseButton1Click:Connect(function()

	ESP = not ESP

	ESPButton.Text =
		ESP and "PLAYER ESP : ON" or "PLAYER ESP : OFF"

	if ESP then

		for _,p in ipairs(Players:GetPlayers()) do
			addESP(p)
		end

	else

		for p in pairs(ESPObjects) do
			removeESP(p)
		end
	end
end)

local function checkKey()

	local entered =
		KeyBox.Text:gsub("%s+",""):lower()

	if entered == KEY then

		KeyStatus.Text = "STATUS: ACCESS GRANTED"
		KeyStatus.TextColor3 = GREEN

		task.wait(.3)

		KeyFrame.Visible = false
		Main.Visible = true

	else

		KeyStatus.Text = "STATUS: INVALID KEY"
		KeyStatus.TextColor3 = RED

		task.wait(.6)

		KeyStatus.Text = "STATUS: WAITING"
		KeyStatus.TextColor3 = GREY
	end
end

Access.MouseButton1Click:Connect(checkKey)

KeyBox.FocusLost:Connect(function(enterPressed)

	if enterPressed then
		checkKey()
	end
end)
